MCP Server Implementation

mcp_server.py 
from fastapi import FastAPI
from pydantic import BaseModel
from typing import List
import numpy as np
from sentence_transformers import SentenceTransformer
from transformers import pipeline

app = FastAPI()

# Load models
embedding_model = SentenceTransformer('all-MiniLM-L6-v2')
generation_model = pipeline("text-generation", model="gpt2")

# Mock vector database
class VectorDB:
    def __init__(self):
        self.documents = [
            "RAG systems combine retrieval with generation.",
            "MCP stands for Model-Client-Proxy architecture.",
            "Vector databases store embeddings for similarity search."
        ]
        self.embeddings = embedding_model.encode(self.documents)
    
    def search(self, query_embedding, top_k=2):
        similarities = np.dot(self.embeddings, query_embedding)
        top_indices = np.argsort(similarities)[-top_k:][::-1]
        return [self.documents[i] for i in top_indices]

vector_db = VectorDB()

class QueryRequest(BaseModel):
    text: str

@app.post("/process")
async def process_query(request: QueryRequest):
    # Step 1: Embed the query
    query_embedding = embedding_model.encode(request.text)
    
    # Step 2: Retrieve relevant documents
    retrieved_docs = vector_db.search(query_embedding)
    
    # Step 3: Generate response using retrieved context
    context = " ".join(retrieved_docs)
    prompt = f"Context: {context}\nQuestion: {request.text}\nAnswer:"
    generated = generation_model(prompt, max_length=100, do_sample=True)
    
    return {
        "query": request.text,
        "retrieved_documents": retrieved_docs,
        "response": generated[0]['generated_text']
    }
client.py 
import requests
import json

class RAGClient:
    def __init__(self, server_url="http://localhost:8080"):
        self.server_url = server_url
    
    def query(self, text):
        try:
            response = requests.post(
                f"{self.server_url}/process",
                json={"text": text},
                headers={"Content-Type": "application/json"}
            )
            response.raise_for_status()
            return response.json()
        except requests.exceptions.RequestException as e:
            return {"error": str(e)}

# Example usage
if __name__ == "__main__":
    client = RAGClient()
    result = client.query("What is RAG?")
    print(json.dumps(result, indent=2))
